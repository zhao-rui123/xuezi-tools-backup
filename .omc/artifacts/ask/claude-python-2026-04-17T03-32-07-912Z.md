# ✅ 用局部变量缓存全局查找
import math
def compute():
    m = math.sqrt  # 缓存到局部
    return m(2) + m(3)

# ✅ 用 @lru_cache 缓存重复调用
from functools import lru_cache

@lru_cache(maxsize=None)
def fib(n):
    return n if n < 2 else fib(n-1) + fib(n-2)

# ✅ 避免重复计算（pandas/numpy 向量化）
import numpy as np
arr = np.array([1, 2, 3])
result = arr ** 2  # 向量化 vs 逐元素 Python 循环
